<refactor_constants_and_adaptive_search>
This edit refactors the scheduling constants, moving them outside the `schedule_job_online` function to be globally accessible. It then implements two key recommendations:
1.  **Adaptive Local Search Window**: The `LOCAL_SEARCH_WINDOW_HOURS` now dynamically expands for high-priority jobs or jobs close to their deadlines, allowing the scheduler to search further for optimal local slots.
2.  **Congestion-Responsive Dynamic Cloud Deterrence**: A new function `_calculate_congestion_factor` is introduced to assess projected local machine utilization. If local machines for a given server type are anticipated to be congested, the `CLOUD_DETERRENCE_PER_HOUR` is dynamically reduced for cloud options of that type, encouraging offloading to the cloud to prevent bottlenecks. The `_calculate_placement_score` function is updated to receive and apply this dynamic cloud deterrence.

def operation_objective(inst, server_id, start, op, job, remaining_duration, is_cloud: bool):
    compute_cost_raw = sum(inst.server_cost[server_id][t % inst.time_horizon]
                           for t in range(start, start + op.duration))

    projected_completion = start + op.duration + remaining_duration
    delay = max(0, projected_completion - job.deadline)

    priority_multiplier = PRIORITY_WEIGHTS.get(job.priority, 1.0) # Default to 1.0 for unknown priorities

    # Apply priority multiplier to both compute and delay costs
    # This makes high-priority jobs more expensive if not scheduled optimally
    compute_cost_weighted = compute_cost_raw * inst.weights['compute_cost'] * priority_multiplier
    delay_penalty_weighted = delay * job.delay_weight * inst.weights['delay_penalty'] * priority_multiplier

    total_cost = compute_cost_weighted + delay_penalty_weighted

    # Add cloud deterrence if using a cloud server, scaled by priority
    if is_cloud:
        total_cost += CLOUD_DETERRENCE_PER_HOUR * op.duration * inst.weights['compute_cost'] * priority_multiplier

    return total_cost

# --- Tunable Parameters ---
# How far into the future local machines search for an optimal slot (in hours).
LOCAL_SEARCH_WINDOW_BASE = 24 # Base search window for local slots

# Dynamic window factors
WINDOW_EXPANSION_CRITICAL_PRIORITY = 1.5 # e.g., for 'emergency', window becomes 24 * 1.5 = 36 hours
WINDOW_EXPANSION_NEAR_DEADLINE = 1.2    # e.g., if deadline is near, window becomes 24 * 1.2 = 28.8 hours
DEADLINE_PROXIMITY_THRESHOLD = 12       # If (deadline - current_time) < this, consider deadline near

# Priority multipliers for delay penalties: higher values make delays more costly for critical jobs.
PRIORITY_WEIGHTS = {
    "emergency": 15.0, # Increased significantly for highest urgency
    "urgent": 8.0,     # Added 'urgent' as a distinct priority, tuned up
    "high": 3.0,
    "medium": 1.0,
    "low": 0.5,
}

# Base cost per unit of duration to deter cloud usage, making local preferred.
CLOUD_DETERRENCE_PER_HOUR = 30.0 # Increased for stronger preference for local

# Reduction factor for cloud deterrence specifically for emergency/urgent jobs.
CLOUD_URGENCY_DISCOUNT_FACTOR = 0.7 # Cloud deterrence is 70% cheaper (1.0 - 0.7 = 0.3 factor applied) for 'emergency' or 'urgent'

# If a high-priority job's best local option causes this much delay (or more),
# consider cloud with further reduced deterrence.
URGENCY_CLOUD_TRIGGER_DELAY_THRESHOLD = 5 # hours of delay

# Additional cost reduction factor for cloud if triggered by urgency threshold.
URGENCY_CLOUD_COST_REDUCTION_FACTOR = 0.5 # Cloud deterrence is halved (factor applied on top of base urgency discount)

# Congestion-responsive cloud deterrence parameters
CONGESTION_LOOKAHEAD_HOURS = 5 # How many hours into the future to check for congestion
CONGESTION_UTILIZATION_THRESHOLD = 0.8 # If avg utilization for a local server type exceeds this, it's congested
CLOUD_CONGESTION_DISCOUNT_FACTOR = 0.6 # If congested, cloud deterrence is reduced by this factor (e.g., 0.6 means 40% discount)

# Tie-breaker for identical scores: prefer server with lower future utilization (to balance load)
LOAD_BALANCING_TIE_BREAKER_FACTOR = 0.00001


def _calculate_congestion_factor(inst: Instance, capacity_usage: Dict[str, Dict[int, int]], server_type: str, current_time: int) -> float:
    """
    Calculates a congestion factor for a given server type.
    Returns 1.0 if not congested, or CLOUD_CONGESTION_DISCOUNT_FACTOR if congested.
    """
    local_servers_of_type = [s for s in inst.servers.values()
                              if not s.is_cloud and s.server_type == server_type]

    if not local_servers_of_type:
        return 1.0 # No local servers, so no local congestion to worry about

    total_capacity_in_window = 0
    total_used_in_window = 0

    for s in local_servers_of_type:
        for t in range(current_time, current_time + CONGESTION_LOOKAHEAD_HOURS):
            total_capacity_in_window += get_available_capacity(inst, s.id, t)
            total_used_in_window += capacity_usage[s.id].get(t, 0)

    if total_capacity_in_window == 0:
        return 1.0 # No capacity, effectively fully utilized, but we can't be more congested. Or no servers.

    avg_utilization = total_used_in_window / total_capacity_in_window

    if avg_utilization > CONGESTION_UTILIZATION_THRESHOLD:
        return CLOUD_CONGESTION_DISCOUNT_FACTOR
    return 1.0


def operation_objective(inst, server_id, start, op, job, remaining_duration, is_cloud: bool, cloud_deterrence_override_factor: float = 1.0):
    compute_cost_raw = sum(inst.server_cost[server_id][t % inst.time_horizon]
                           for t in range(start, start + op.duration))

    projected_completion = start + op.duration + remaining_duration
    delay = max(0, projected_completion - job.deadline)

    priority_multiplier = PRIORITY_WEIGHTS.get(job.priority, 1.0) # Default to 1.0 for unknown priorities

    # Apply priority multiplier to both compute and delay costs
    compute_cost_weighted = compute_cost_raw * inst.weights['compute_cost'] * priority_multiplier
    delay_penalty_weighted = delay * job.delay_weight * inst.weights['delay_penalty'] * priority_multiplier

    total_cost = compute_cost_weighted + delay_penalty_weighted

    # Add cloud deterrence if using a cloud server, applying dynamic override
    if is_cloud:
        deterrence = CLOUD_DETERRENCE_PER_HOUR * op.duration * inst.weights['compute_cost'] * priority_multiplier
        # Apply the override factor (e.g., from emergency or congestion discounts)
        deterrence *= cloud_deterrence_override_factor
        total_cost += deterrence

    # Tie-breaker: prefer earlier starts slightly if costs are otherwise equal.
    # A small coefficient to ensure it only breaks ties, not overrides actual costs.
    total_cost += start * 0.0001

    return total_cost
