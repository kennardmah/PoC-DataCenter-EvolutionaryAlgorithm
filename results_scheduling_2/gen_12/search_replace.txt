def schedule_job_online(inst: Instance,
                        job: Job,
                        schedule: Schedule,
                        capacity_usage: Dict[str, Dict[int, int]],
                        current_time: int):

    # Define parameters for the scheduling heuristic
    PRIORITY_MULTIPLIERS = {
        'emergency': 5.0, # Highest priority, severe penalty for delay
        'high': 2.0,
        'medium': 1.0,
        'low': 0.5,       # Lower priority, more flexible with delay
    }
    CLOUD_DETERRENCE_PENALTY = 10.0 # A configurable penalty to discourage cloud usage
    MAX_LOCAL_SEARCH_HORIZON = inst.time_horizon * 2 # Search up to 2 full cycles for local machines

    # Helper function to calculate the score for a proposed operation placement
    def _calculate_placement_score(
        _inst: Instance,
        _job: Job,
        _operation: Operation,
        _proposed_start: int,
        _server_id: str,
        _remaining_ops_duration: int,
        _priority_multipliers: Dict[str, float],
        _cloud_deterrence_penalty: float
    ) -> float:
        """
        Calculates a weighted score for a proposed operation placement,
        considering compute cost and the impact on the job's overall delay penalty.
        """
        # Compute cost for THIS operation
        op_compute_cost = sum(_inst.server_cost[_server_id][t % _inst.time_horizon]
                              for t in range(_proposed_start, _proposed_start + _operation.duration))

        # Projected completion time for THE ENTIRE JOB
        projected_op_end = _proposed_start + _operation.duration
        projected_job_completion = projected_op_end + _remaining_ops_duration

        # Delay penalty for THE ENTIRE JOB
        delay = max(0, projected_job_completion - _job.deadline)

        # Apply priority multiplier to delay penalty
        priority_factor = _priority_multipliers.get(_job.priority, 1.0)
        delay_penalty = delay * _job.delay_weight * priority_factor * _inst.weights['delay_penalty']

        # Total score for this specific operation placement
        total_score = (
            op_compute_cost * _inst.weights['compute_cost']
            + delay_penalty
        )

        # Add cloud deterrence penalty
        if _inst.servers[_server_id].is_cloud:
            total_score += _cloud_deterrence_penalty

        # Tie-breaker: prefer earlier starts slightly if costs are otherwise equal
        total_score += _proposed_start * 0.001

        return total_score

    schedule[job.id] = []
    # current_start tracks the earliest an operation can start based on previous operations in this job
    op_start_time_predecessor = max(job.arrival_time, current_time)

    for idx, op in enumerate(job.operations):
        remaining_operations_duration = sum(o.duration for o in job.operations[idx + 1:])

        best_score = float('inf')
        chosen_server_id = None
        chosen_start_time = None

        # --- Evaluate Cloud Options First ---
        # Cloud servers are always immediately available and have unlimited capacity
        for server_id, server in inst.servers.items():

                proposed_start = op_start_time_predecessor

                score = _calculate_placement_score(
                    inst, job, op, proposed_start, server_id, remaining_operations_duration, PRIORITY_MULTIPLIERS, CLOUD_DETERRENCE_PENALTY
                )

                if score < best_score:
                    best_score = score
                    chosen_server_id = server_id
                    chosen_start_time = proposed_start
                # No need to search further in time for cloud, it's always earliest
                break # Only one cloud server per type usually, so we pick the first one

        # --- Evaluate Local Options ---
        for server_id, server in inst.servers.items():
            if not server.is_cloud and server.server_type == op.server_type:
                # Search for the best available slot on this local server within a reasonable horizon
                # Search from the earliest possible start up to a future horizon (e.g., job deadline + 24 hours, or a fixed window)
                search_end_time = min(job.deadline + inst.time_horizon, op_start_time_predecessor + MAX_LOCAL_SEARCH_HORIZON)

                for proposed_start in range(op_start_time_predecessor, search_end_time + 1):
                    if has_capacity(inst, capacity_usage, server_id, proposed_start, op.duration, op.capacity_required):

                        )

                        if score < best_score:
                            best_score = score
                            chosen_server_id = server_id
                            chosen_start_time = proposed_start
                        # Continue searching to find the *best* local slot, not just the first

        # If after checking both cloud and local, no option was found (shouldn't happen with cloud fallback)
        if chosen_server_id is None:
            # This indicates a logical error or extreme scenario where even cloud fails, fallback to cloud immediately
            # This should ideally not be reached if cloud is truly unlimited.

            # for the current operation at op_start_time_predecessor

                    chosen_start_time = op_start_time_predecessor
                    break
            if chosen_server_id is None: # Still no cloud, critical error
