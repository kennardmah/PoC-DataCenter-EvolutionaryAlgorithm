#   - Use global knowledge about all jobs

# Constants for the scheduler (can be tuned)
PRIORITY_WEIGHTS = {
    "emergency": 8.0,  # Highest urgency, highest penalty for delay
    "high": 3.0,
    "medium": 1.0,
    "low": 0.5,
}
# A fixed penalty per hour of cloud usage, scaled by compute_cost_weight and priority
CLOUD_DETERRENCE_PER_HOUR = 0.5
# How far into the future local machines search for an optimal slot (in hours).
LOCAL_SEARCH_WINDOW_HOURS = 24


def _calculate_placement_score(inst: Instance, server_id: str, start: int, op: Operation, job: Job, remaining_duration: int, is_cloud: bool) -> float:
    """
    Calculates a weighted score for a proposed operation placement,
    considering compute cost, the impact on the job's overall delay penalty,
    and dynamic cloud deterrence based on priority.
    """
    # Compute cost for THIS operation
    op_compute_cost_raw = sum(inst.server_cost[server_id][t % inst.time_horizon]
                              for t in range(start, start + op.duration))

    # Projected completion time for THE ENTIRE JOB
    projected_op_end = start + op.duration
    projected_job_completion = projected_op_end + remaining_duration

    # Calculate the delay penalty for THE ENTIRE JOB.
    delay = max(0, projected_job_completion - job.deadline)

    # Apply priority multiplier to the delay penalty.
    priority_factor = PRIORITY_WEIGHTS.get(job.priority, 1.0)
    delay_penalty = delay * job.delay_weight * priority_factor * inst.weights['delay_penalty']

    # Apply priority multiplier to compute cost as well
    compute_cost_weighted = op_compute_cost_raw * inst.weights['compute_cost'] * priority_factor

    total_cost = compute_cost_weighted + delay_penalty

    # Add cloud deterrence if using a cloud server, scaled by priority
    if is_cloud:
        total_cost += CLOUD_DETERRENCE_PER_HOUR * op.duration * inst.weights['compute_cost'] * priority_factor

    # Tie-breaker: prefer earlier starts slightly if costs are otherwise equal
    total_cost += start * 0.0001

    return total_cost


def schedule_job_online(inst: Instance,
                        job: Job,
                        schedule: Schedule,
                        capacity_usage: Dict[str, Dict[int, int]],
                        current_time: int):

    schedule[job.id] = []
    # op_start_time_for_job tracks the earliest an operation of THIS job can start,
    # considering its arrival time and the completion of its previous operation.
    op_start_time_for_job = max(job.arrival_time, current_time)

    for idx, op in enumerate(job.operations):
        # Remaining duration includes subsequent operations of this job to project total job completion time
        remaining_duration = sum(o.duration for o in job.operations[idx + 1:])

        # Initialize best option for this operation.
        # Tuple format: (total_objective_cost, projected_start_time, server_id, is_cloud_choice)
        # We use a very high initial cost and late start to ensure any valid option is better.
        best_op_candidate = (float('inf'), float('inf'), "", True)

        # --- Evaluate Cloud Options First (immediate availability) ---
        for server_id, server in inst.servers.items():
            if server.is_cloud and server.server_type == op.server_type:
                # Cloud servers: always available immediately at op_start_time_for_job
                cost = _calculate_placement_score(inst, server_id, op_start_time_for_job, op, job, remaining_duration, is_cloud=True)
                best_op_candidate = min(best_op_candidate, (cost, op_start_time_for_job, server_id, True))
                # For cloud, we typically pick the cheapest/first one of compatible type since capacity is infinite and start time is fixed.
                # If there are multiple cloud servers of the same type with different costs, the min will handle it.

        # --- Evaluate Local Options (best fit within window for EACH local server) ---
        # The horizon extends up to the job's deadline plus one time_horizon to allow for some flexibility.
        search_end_time = job.deadline + inst.time_horizon # Max search window for local servers

        for server_id, server in inst.servers.items():
            if not server.is_cloud and server.server_type == op.server_type:
                # Track the best option found *for this specific local server* within its search window
                best_for_this_local_server = (float('inf'), float('inf'), "") # (cost, start, server_id)

                # Search within a dynamically determined window: from op_start_time_for_job up to search_end_time
                for start in range(op_start_time_for_job, search_end_time + 1):
                    if has_capacity(inst, capacity_usage, server_id, start, op.duration, op.capacity_required):
                        cost = _calculate_placement_score(inst, server_id, start, op, job, remaining_duration, is_cloud=False)
                        # Update the best option for this local server if the current slot is better
                        best_for_this_local_server = min(best_for_this_local_server, (cost, start, server_id))
                        # CRITICAL: DO NOT break. Continue searching for a better slot on the *same* local server
                        # within the search window to find the BEST fit.

                # After checking all possible slots on this local server, if a valid slot was found,
                # compare it with the current best_op_candidate (which includes cloud and other local servers).
                if best_for_this_local_server[2] != "": # Check if a valid local slot was found
                    local_cost, local_start, local_id = best_for_this_local_server
                    best_op_candidate = min(best_op_candidate, (local_cost, local_start, local_id, False))

        # Unpack the chosen best option
        chosen_cost, chosen_start, chosen_server, _ = best_op_candidate

        # Error handling if no suitable slot found (should not happen if cloud is always an option).
        if chosen_server is None or chosen_cost == float('inf'):
            # Fallback for robustness, although it indicates a severe configuration issue
            # if no cloud server or local slot was found.
            raise RuntimeError(f"Could not schedule operation {idx} for job {job.id}: No compatible server (local or cloud) found after exhaustive search or search horizon was too short.")


        # Reserve capacity for the chosen operation in the `capacity_usage` dictionary.
        for t in range(chosen_start, chosen_start + op.duration):
            capacity_usage[chosen_server][t] = capacity_usage[chosen_server].get(t, 0) + op.capacity_required

        # Add the scheduled operation to the job's schedule.
        schedule[job.id].append({"server": chosen_server, "start": chosen_start})

        # Update op_start_time_for_job for the next operation of this job
        # It must start after the current operation finishes.
        op_start_time_for_job = chosen_start + op.duration
