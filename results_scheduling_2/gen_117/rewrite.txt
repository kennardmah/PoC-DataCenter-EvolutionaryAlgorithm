TIME_HORIZON = 24

@dataclass
class Server:
    id: str
    server_type: str
    capacity: int
    is_cloud: bool = False


@dataclass
class Operation:
    server_type: str
    duration: int
    capacity_required: int = 1


@dataclass
class Job:
    id: str
    description: str
    priority: str
    deadline: int
    delay_weight: float
    operations: List[Operation]
    arrival_time: int = 0


@dataclass
class MachineFailure:
    machine_id: str
    failure_start: int
    failure_end: int
    capacity_reduction: int
    failure_rate: float = 1.0
    recovery_rate: float = 1.0
    reason: str = ""
    is_random: bool = False


@dataclass
class Instance:
    servers: Dict[str, Server]
    jobs: Dict[str, Job]
    server_cost: Dict[str, Dict[int, float]]
    time_horizon: int
    weights: Dict[str, float]
    machine_failures: Optional[List[MachineFailure]] = None


Schedule = Dict[str, List[Dict[str, Any]]]

    failure_scenario: Optional[str] = None,
    enable_random_failures: bool = True,
    random_failure_rate: float = 6.0,
    random_seed: Optional[int] = 42
) -> Instance:
    """
    Create the problem instance from CSV files.

    Args:
        scenario: Job scenario to load from CSV
        failure_scenario: Machine failure scenario or None for no failures
        enable_random_failures: If True, add stochastic background failures
        random_failure_rate: Mean hours between random failures per machine
        random_seed: Random seed for reproducible failures (None for true randomness)

    Returns:
        Instance object with jobs, servers, and optional machine failures from CSV
    """
    from helpers.scenario_loader import create_instance_from_csv
    return create_instance_from_csv(
        scenario=scenario,
        failure_scenario=failure_scenario,
        enable_random_failures=enable_random_failures,
        random_failure_rate=random_failure_rate,
        random_seed=random_seed
    )

@dataclass
class Fitness:
    score: float
    compute_cost: float
    delay_penalty: float
    violations: List[str]
    job_completion_times: Dict[str, int]
    cloud_relocations: List[str] = None

    def __post_init__(self):
        if self.cloud_relocations is None:
            self.cloud_relocations = []

    def is_valid(self) -> bool:
        return len(self.violations) == 0


def evaluate(inst: Instance, schedule: Schedule) -> Fitness:
    """
    Evaluate a schedule with capacity constraints.
    """
    violations = []
    total_compute_cost = 0
    total_delay_penalty = 0
    job_completion_times = {}
    cloud_relocations = []

    # Track capacity usage per machine per timestep
    capacity_usage = {s: {} for s in inst.servers}

    def get_available_capacity_at_time(server_id: str, time: int) -> int:
        """Calculate available capacity considering exponential failure/recovery."""
        base_capacity = inst.servers[server_id].capacity

        if not inst.machine_failures:
            return base_capacity

        total_reduction = 0
        for failure in inst.machine_failures:
            if failure.machine_id != server_id:
                continue

            if time < failure.failure_start:
                continue
            elif time >= failure.failure_end:
                time_since_end = time - failure.failure_end
                recovery_factor = 1.0 - math.exp(-failure.recovery_rate * time_since_end)
                reduction = failure.capacity_reduction * (1.0 - recovery_factor)
                total_reduction += reduction
            else:
                time_since_start = time - failure.failure_start
                failure_factor = 1.0 - math.exp(-failure.failure_rate * time_since_start)
                reduction = failure.capacity_reduction * failure_factor
                total_reduction += reduction

        return max(0, base_capacity - int(total_reduction))

    # Process each job's operations
    for job_id, ops in schedule.items():
        if job_id not in inst.jobs:
            violations.append(f"Unknown job {job_id}")
            continue

        job = inst.jobs[job_id]
        current_time = 0

        for idx, op_sched in enumerate(ops):
            if idx >= len(job.operations):
                violations.append(f"{job_id}: too many operations")
                continue

            operation = job.operations[idx]
            server_id = op_sched["server"]
            start = op_sched["start"]
            duration = operation.duration
            capacity_req = operation.capacity_required

            # Check server exists and type matches
            if server_id not in inst.servers:
                violations.append(f"{job_id} op {idx}: unknown server {server_id}")
                continue

            server = inst.servers[server_id]
            if server.server_type != operation.server_type:
                violations.append(
                    f"{job_id} op {idx}: type mismatch ({server.server_type} != {operation.server_type})"
                )
                continue

            # Check precedence
            if start < current_time:
                violations.append(f"{job_id} op {idx}: starts at {start} before previous ends at {current_time}")

            current_time = start + duration

            # Calculate compute cost
            for t in range(start, start + duration):
                hour = t % inst.time_horizon
                cost_per_hour = inst.server_cost[server_id][hour]
                total_compute_cost += cost_per_hour

            # Track capacity usage
            for t in range(start, start + duration):
                if t not in capacity_usage[server_id]:
                    capacity_usage[server_id][t] = 0
                capacity_usage[server_id][t] += capacity_req

        # Record job completion time
        job_completion_times[job_id] = current_time

        # Calculate delay penalty
        delay = max(0, current_time - job.deadline)
        total_delay_penalty += delay * job.delay_weight

    # Check capacity constraints
    for server_id, usage_by_time in capacity_usage.items():
        server = inst.servers[server_id]
        if server.is_cloud:
            continue  # Cloud has unlimited capacity

        for time, used in usage_by_time.items():
            available = get_available_capacity_at_time(server_id, time)
            if used > available:
                violations.append(
                    f"{server_id}: capacity exceeded at t={time} (used: {used}, available: {available})"
                )

    # Calculate weighted score
    score = (
        total_compute_cost * inst.weights["compute_cost"]
        + total_delay_penalty * inst.weights["delay_penalty"]
    )

    # Add penalty for violations
    if violations:
        score += 1e6 * len(violations)

    return Fitness(
        score=score,
        compute_cost=total_compute_cost,
        delay_penalty=total_delay_penalty,
        violations=violations,
        job_completion_times=job_completion_times,
        cloud_relocations=cloud_relocations,
    )

@dataclass
class SchedulerContext:
    inst: Instance
    schedule: Schedule
    capacity_usage: Dict[str, Dict[int, int]] = field(default_factory=lambda: defaultdict(dict))

    # --- Tunable Parameters ---
    PRIORITY_WEIGHTS: Dict[str, float] = field(default_factory=lambda: {
        "emergency": 8.0,
        "high": 3.0,
        "medium": 1.0,
        "low": 0.5,
    })
    CLOUD_DETERRENCE_PER_HOUR: float = 0.7  # Penalty per hour of cloud usage
    LOCAL_SEARCH_WINDOW_HOURS: int = 24  # Base window for local machine search
    LOCAL_GAP_PENALTY_PER_HOUR: float = 0.2 # Penalty for idle time between ops on local machines
    LOCAL_CONGESTION_THRESHOLD_RATIO: float = 0.5 # If avg local capacity below this, reduce cloud deterrence
    CLOUD_DETERRENCE_REDUCTION_ON_CONGESTION: float = 0.5 # Factor to reduce cloud deterrence by if congested


    def get_available_capacity(self, server_id: str, time: int) -> int:
        """Get available capacity at a specific time (for scheduling decisions)."""
        base_capacity = self.inst.servers[server_id].capacity
        if not self.inst.machine_failures:
            return base_capacity

        total_reduction = 0
        for failure in self.inst.machine_failures:
            if failure.machine_id != server_id:
                continue
            if time < failure.failure_start:
                continue
            elif time >= failure.failure_end:
                time_since_end = time - failure.failure_end
                recovery_factor = 1.0 - math.exp(-failure.recovery_rate * time_since_end)
                reduction = failure.capacity_reduction * (1.0 - recovery_factor)
                total_reduction += reduction
            else:
                time_since_start = time - failure.failure_start
                failure_factor = 1.0 - math.exp(-failure.failure_rate * time_since_start)
                reduction = failure.capacity_reduction * failure_factor
                total_reduction += reduction

        return max(0, base_capacity - int(total_reduction))

    def has_capacity(self, server_id: str, start: int, duration: int, req: int) -> bool:
        """Check if a server has capacity for an operation at a given time window."""
        for t in range(start, start + duration):
            used = self.capacity_usage[server_id].get(t, 0)
            if used + req > self.get_available_capacity(server_id, t):
                return False
        return True

    def _get_avg_local_capacity_for_type(self, server_type: str, current_time: int, duration: int) -> float:
        """Estimates average available capacity for a server type over a short window."""
        total_capacity = 0
        total_available = 0
        server_count = 0
        
        # Consider a small lookahead window to assess current congestion
        check_window = range(current_time, current_time + duration + 1)
        
        for server_id, server in self.inst.servers.items():
            if not server.is_cloud and server.server_type == server_type:
                server_count += 1
                for t in check_window:
                    total_capacity += server.capacity
                    total_available += self.get_available_capacity(server_id, t) - self.capacity_usage[server_id].get(t, 0)
        
        if total_capacity == 0:
            return 1.0 # No local servers of this type, assume not congested
        
        return total_available / total_capacity


@dataclass
class OperationEvaluator:
    context: SchedulerContext
    job: Job
    op_idx: int
    op_start_time_predecessor: int # Earliest start due to previous operations
    remaining_ops_duration: int

    @property
    def operation(self) -> Operation:
        return self.job.operations[self.op_idx]

    def _calculate_placement_score(self, server_id: str, start: int, is_cloud: bool) -> float:
        """
        Calculates a weighted score for a proposed operation placement,
        considering compute cost, delay penalty, and dynamic cloud deterrence.
        """
        op = self.operation
        inst = self.context.inst
        priority_factor = self.context.PRIORITY_WEIGHTS.get(self.job.priority, 1.0)

        # 1. Compute Cost
        op_compute_cost_raw = sum(inst.server_cost[server_id][t % inst.time_horizon]
                                  for t in range(start, start + op.duration))
        compute_cost_weighted = op_compute_cost_raw * inst.weights['compute_cost']

        # 2. Delay Penalty
        projected_op_end = start + op.duration
        projected_job_completion = projected_op_end + self.remaining_ops_duration
        delay = max(0, projected_job_completion - self.job.deadline)
        delay_penalty = delay * self.job.delay_weight * priority_factor * inst.weights['delay_penalty']

        # 3. Cloud Deterrence (Dynamic)
        cloud_deterrence_cost = 0.0
        if is_cloud:
            cloud_deterrence_factor = 1.0
            avg_local_cap_ratio = self.context._get_avg_local_capacity_for_type(op.server_type, start, op.duration)

            if avg_local_cap_ratio < self.context.LOCAL_CONGESTION_THRESHOLD_RATIO:
                # Local machines are congested, reduce cloud deterrence
                cloud_deterrence_factor = 1.0 - self.context.CLOUD_DETERRENCE_REDUCTION_ON_CONGESTION
            
            cloud_deterrence_cost = self.context.CLOUD_DETERRENCE_PER_HOUR * op.duration * inst.weights['compute_cost'] * priority_factor * cloud_deterrence_factor

        # 4. Intra-Job Dependency Cost (for local machines)
        local_gap_penalty = 0.0
        if not is_cloud and start > self.op_start_time_predecessor:
            idle_time = start - self.op_start_time_predecessor
            local_gap_penalty = idle_time * self.context.LOCAL_GAP_PENALTY_PER_HOUR * inst.weights['compute_cost']

        total_cost = compute_cost_weighted + delay_penalty + cloud_deterrence_cost + local_gap_penalty

        # Tie-breaker: prefer earlier starts slightly if costs are otherwise equal.
        # Then prefer local over cloud (False < True).
        # Then prefer smaller server_id (alphabetical)
        return (total_cost, start, is_cloud, server_id) # Return tuple for robust min() comparison


    def find_best_placement(self):
        """
        Finds the best server and start time for the current operation.
        """
        best_overall_candidate = (float('inf'), float('inf'), "", True) # (score, start, server_id, is_cloud)
        
        # Determine effective search window for local servers based on priority (Recommendation 5)
        effective_local_search_window = self.context.LOCAL_SEARCH_WINDOW_HOURS
        if self.job.priority == 'emergency':
            effective_local_search_window += self.context.inst.time_horizon // 2 # Extend for emergency
        elif self.job.priority == 'high':
            effective_local_search_window += self.context.inst.time_horizon // 4 # Slightly extend for high

        # Evaluate all compatible servers (local and cloud)
        for server_id, server in self.context.inst.servers.items():
            if server.server_type != self.operation.server_type:
                continue

            if server.is_cloud:
                proposed_start = self.op_start_time_predecessor
                candidate_score_tuple = self._calculate_placement_score(server_id, proposed_start, True)
                best_overall_candidate = min(best_overall_candidate, candidate_score_tuple)
            else:
                # Local: search for the best available slot within the dynamic search window (Recommendation 1)
                best_local_for_server = (float('inf'), float('inf'), "", False) # (score, start, server_id, is_cloud)

                search_end_time = self.op_start_time_predecessor + effective_local_search_window
                
                # Ensure search doesn't go unnecessarily far beyond deadline + horizon
                search_end_time = min(search_end_time, self.job.deadline + self.context.inst.time_horizon)

                for start in range(self.op_start_time_predecessor, search_end_time + 1):
                    if self.context.has_capacity(server_id, start, self.operation.duration, self.operation.capacity_required):
                        candidate_score_tuple = self._calculate_placement_score(server_id, start, False)
                        # Keep the best slot found for *this specific local server*
                        best_local_for_server = min(best_local_for_server, candidate_score_tuple)
                        # Do NOT break here. We want the BEST slot on this server, not the first.

                # If a valid slot was found on this local server, compare it with the overall best.
                if best_local_for_server[0] != float('inf'):
                    best_overall_candidate = min(best_overall_candidate, best_local_for_server)

        # Extract the chosen start time and server ID
        _, chosen_start, chosen_is_cloud, chosen_server_id = best_overall_candidate
        
        if chosen_server_id == "": # No server was found, this indicates a problem
            raise RuntimeError(f"No suitable server found for operation {self.op_idx} of job {self.job.id} at time {self.op_start_time_predecessor}. Check instance configuration or search parameters.")

        return chosen_server_id, chosen_start


def schedule_job_online(inst: Instance,
                        job: Job,
                        schedule: Schedule,
                        capacity_usage: Dict[str, Dict[int, int]],
                        current_time: int):

    context = SchedulerContext(inst=inst, schedule=schedule, capacity_usage=capacity_usage)
    schedule[job.id] = []
    
    # op_start_time_for_job tracks the earliest an operation of THIS job can start,
    # considering its arrival time and the completion of its previous operation.
    op_start_time_for_job = max(job.arrival_time, current_time)

    for idx, op in enumerate(job.operations):
        # Remaining duration includes subsequent operations of this job to project total job completion time
        remaining_operations_duration = sum(o.duration for o in job.operations[idx + 1:])

        evaluator = OperationEvaluator(
            context=context,
            job=job,
            op_idx=idx,
            op_start_time_predecessor=op_start_time_for_job,
            remaining_ops_duration=remaining_operations_duration
        )

        chosen_server_id, chosen_start_time = evaluator.find_best_placement()

        # Reserve capacity for the chosen slot
        for t in range(chosen_start_time, chosen_start_time + op.duration):
            context.capacity_usage[chosen_server_id][t] = context.capacity_usage[chosen_server_id].get(t, 0) + op.capacity_required

        # Add the scheduled operation to the job's schedule.
        schedule[job.id].append({"server": chosen_server_id, "start": chosen_start_time})

        # Update op_start_time_for_job for the next operation of this job
        # It must start after the current operation finishes.
        op_start_time_for_job = chosen_start_time + op.duration

# EVENT-DRIVEN ONLINE EXECUTION LOOP (NOT EVOLVABLE - ENFORCES ONLINE CONSTRAINT)

def construct_schedule(inst: Instance) -> Schedule:
    """
    STRICT ONLINE SCHEDULER - NO FUTURE KNOWLEDGE

    This function enforces the online constraint and is NOT evolved.
    Only schedule_job_online (above) can be modified during evolution.

    CRITICAL ONLINE CONSTRAINT:
    When scheduling a job arriving at time T, you can ONLY access:
    - Jobs that have already been scheduled (arrival_time < T)
    - The current job (arrival_time == T)

    You CANNOT access or analyze jobs with arrival_time > T

    Implementation: Process jobs in arrival order, one at a time.
    """
    schedule = {}
    capacity_usage = defaultdict(dict)

    # Create list of (arrival_time, job) tuples and sort ONLY by arrival time
    # Note: We must access inst.jobs to know when they arrive, but we
    # process them strictly in order without lookahead
    job_arrivals = [(job.arrival_time, job) for job in inst.jobs.values()]
    job_arrivals.sort(key=lambda x: x[0])  # Sort only by arrival time

    # Process each job as it arrives (strict temporal order)
    for arrival_time, job in job_arrivals:
        # At this moment, only this job and earlier jobs exist in our world
        # Jobs arriving later are "in the future" and unknown
        schedule_job_online(inst, job, schedule, capacity_usage, arrival_time)

    return schedule


# =============================================================================
# RUN SCHEDULING (ENTRY POINT FOR EVALUATION)
# =============================================================================

def run_scheduling(
    scenario: str = "baseline",
    failure_scenario: Optional[str] = None,
    multi_scenario: bool = False,
    scenarios: Optional[List[str]] = None,
    scenario_weights: Optional[Dict[str, float]] = None,
    failure_scenarios: Optional[Dict[str, str]] = None
):
    """
    Run the schedule constructor and return results.

    Args:
        scenario: Job scenario to load from CSV (for single-scenario mode)
        failure_scenario: Machine failure scenario from CSV or None (for single-scenario mode)
        multi_scenario: If True, evaluate across multiple scenarios
        scenarios: List of scenario names to evaluate (for multi-scenario mode)
        scenario_weights: Weight for each scenario in final score (for multi-scenario mode)
        failure_scenarios: Mapping of scenario name to failure scenario (for multi-scenario mode)
    """
    if multi_scenario and scenarios:
        # Multi-scenario evaluation
        results = {}
        weighted_score = 0.0

        for scenario_name in scenarios:
            # Get failure scenario for this scenario (if any)
            fail_scenario = None
            if failure_scenarios and scenario_name in failure_scenarios:
                fail_scenario = failure_scenarios[scenario_name]

            # Create instance and run scheduling
            inst = create_instance(scenario=scenario_name, failure_scenario=fail_scenario)
            schedule = construct_schedule(inst)
            fitness = evaluate(inst, schedule)
            score = float(fitness.score)

            # Store results
            results[scenario_name] = (schedule, fitness, score)

            # Add weighted contribution
            weight = scenario_weights.get(scenario_name, 1.0) if scenario_weights else 1.0
            weighted_score += weight * score

        # Return multi-scenario results
        return {
            "scenarios": results,
            "weighted_score": weighted_score,
            "weights": scenario_weights or {s: 1.0 for s in scenarios}
        }
    else:
        # Single scenario evaluation (backward compatibility)
        inst = create_instance(scenario=scenario, failure_scenario=failure_scenario)
        schedule = construct_schedule(inst)
        fitness = evaluate(inst, schedule)
        return schedule, fitness, float(fitness.score)
