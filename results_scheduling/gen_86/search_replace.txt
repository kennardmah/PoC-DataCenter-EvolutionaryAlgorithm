        def _calculate_operation_objective_value(
            self, server_id: str, start: int, op: Operation, job: Job,
            remaining_operations_duration: int
        ) -> float:
            """
            Calculates a combined objective value for an operation based on weighted compute cost
            and projected delay penalty for the entire job.
            """
            # 1. Compute Cost for this operation
            compute_cost_op = self._calculate_operation_compute_cost(server_id, start, op.duration)
            weighted_compute_cost = compute_cost_op * self.inst.weights["compute_cost"]

            # 2. Projected Delay Penalty for the job
            current_op_finish_time = start + op.duration
            projected_job_completion_time = current_op_finish_time + remaining_operations_duration

            projected_delay = max(0, projected_job_completion_time - job.deadline)
            weighted_delay_penalty = projected_delay * job.delay_weight * self.inst.weights["delay_penalty"]

            # Total objective for this specific operation placement
            return weighted_compute_cost + weighted_delay_penalty
