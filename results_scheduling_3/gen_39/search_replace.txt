# Heuristic parameters
PRIORITY_FACTORS = {
    "Emergency": 3.0,  # Higher penalty for emergency jobs
    "High": 2.0,       # High penalty for high priority
    "Medium": 1.0,     # Default penalty
    "Low": 0.5         # Lower penalty for low priority
}
CLOUD_COST_MULTIPLIER = 1.65 # Make cloud machines more expensive to further encourage local usage

# Procrastination penalty factors based on priority
# Higher factor means more penalty for waiting
PROCR_PENALTY_FACTORS = {
    "Emergency": 0.5, # Very high penalty for Emergency jobs to start late
    "High": 0.2,      # High penalty for High priority jobs
    "Medium": 0.05,   # Moderate penalty for Medium jobs
    "Low": 0.01       # Low penalty for Low priority jobs
}

# Base search horizons for local machines, adjusted dynamically based on slack
BASE_LOCAL_SEARCH_HORIZON = {
    "Emergency": 24, # Critical: focus on immediate placement
    "High": 48,      # Urgent: reasonable horizon
    "Medium": 72,    # Normal: wider search for optimization
    "Low": 96        # Flexible: widest search
}
# Absolute maximum search window to prevent excessive computation for very long deadlines
MAX_ABSOLUTE_SEARCH_WINDOW = 3 * TIME_HORIZON # e.g., 3 days ahead

def operation_objective_combined(inst, server_id, start, op, job, remaining_duration, is_cloud, op_release_time):
    """
    Calculates the objective cost for a given operation, incorporating priority-based
    delay penalties, cloud cost multiplier, and a procrastination penalty.
    """
    compute = sum(inst.server_cost[server_id][t % inst.time_horizon]
                  for t in range(start, start + op.duration))

    if is_cloud:
        compute *= CLOUD_COST_MULTIPLIER

    projected_completion = start + op.duration + remaining_duration
    delay = max(0, projected_completion - job.deadline)

    # Apply priority factor to delay penalty
    priority_factor = PRIORITY_FACTORS.get(job.priority, 1.0)
    delay_penalty = delay * job.delay_weight * priority_factor

    # Procrastination Penalty: Penalize waiting time, scaled by job importance.
    # The wait_time is the difference between actual start and the earliest possible start.
    wait_time = max(0, start - op_release_time)
    procrastination_factor = PROCR_PENALTY_FACTORS.get(job.priority, 0.05)
    procrastination_penalty = wait_time * job.delay_weight * procrastination_factor

    return (
        compute * inst.weights['compute_cost']
        + delay_penalty * inst.weights['delay_penalty']
        + procrastination_penalty # Add this new component
    )

def has_capacity_optimized(capacity_usage, server_id, start, duration, req, cached_server_capacity):
    """
    Optimized capacity check using a pre-computed/cached available capacity for the current server.
    cached_server_capacity: dict {time: effective_capacity} for the specific server.
    """
    for t in range(start, start + duration):
        used = capacity_usage[server_id].get(t, 0)
        # Use cached available capacity if possible
        available = cached_server_capacity.get(t, 0) # Fallback to 0 if not pre-cached (should be pre-cached though)
        if used + req > available:
            return False
    return True

        if cloud_server_candidates:
            # Find the cheapest cloud server for this operation type
            cheapest_cloud_id = min(cloud_server_candidates, key=lambda s_id: inst.server_cost[s_id][current_op_release_time % inst.time_horizon])
            cloud_score = operation_objective_combined(inst, cheapest_cloud_id, current_op_release_time, op, job, remaining_duration, True, current_op_release_time)
            best_placement = (cloud_score, cheapest_cloud_id, current_op_release_time)

        # Slack: time remaining until deadline if scheduled immediately after current_op_release_time

        slack_to_deadline = job.deadline - projected_job_completion_if_scheduled_now

        base_horizon = BASE_LOCAL_SEARCH_HORIZON.get(job.priority, BASE_LOCAL_SEARCH_HORIZON["Medium"])

        if slack_to_deadline <= -op.duration: # Already behind schedule (critical)
            search_horizon_value = base_horizon // 2 # Aggressively search a short window
