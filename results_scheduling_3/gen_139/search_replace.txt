5.  **Refined Cloud vs. Local Decision Logic:** The final decision uses the `cloud_aggressiveness_threshold` to dynamically adjust the preference for local vs. cloud machines based on job priority, allowing urgent jobs to more readily consider cloud resources if local options are not sufficiently superior.

These changes aim to restore proven mechanisms for balancing cost, urgency, and resource contention, which are crucial for high performance in complex online scheduling scenarios.

class PlacementEvaluator:
    """
    Encapsulates the logic for finding the best placement for a single operation
    using a staged-decision heuristic with enhanced penalties and adaptive parameters.
    """
    def __init__(self, inst, job, op, capacity_usage, earliest_start, remaining_duration):
        self.inst = inst
        self.job = job
        self.op = op
        self.capacity_usage = capacity_usage
        self.earliest_start = earliest_start
        self.remaining_duration = remaining_duration

        # --- Heuristic Parameters (Re-introduced and Refined) ---
        # Search horizon: Short for urgent jobs, long for non-urgent.
        self.priority_base_horizon = {
            'emergency': 4, 'critical': 8, 'urgent': 16,
            'high': 32, 'medium': 64, 'low': 96
        }

        # Procrastination penalty: High penalty for delaying important jobs.
        self.priority_procrastination_multiplier = {
            'emergency': 12.0, 'critical': 10.0, 'urgent': 5.0,
            'high': 2.5, 'medium': 1.0, 'low': 0.5
        }
        self.base_procrastination_penalty_factor = 0.075 # Slightly increased base penalty

        # Cloud aggressiveness: High-prio jobs take cloud unless local is significantly better.
        # A value < 1 means cloud is preferred; > 1 means local is preferred (even if slightly costlier).
        self.cloud_aggressiveness_threshold = {
            'emergency': 0.85, 'critical': 0.90, 'urgent': 0.95,
            'high': 1.05, 'medium': 1.15, 'low': 1.25
        }

        # Load penalty: High-prio jobs are very sensitive to server congestion.
        self.load_penalty_factor = 7.5 # Increased factor for more aggressive load balancing
        self.priority_load_multiplier = {
            "emergency": 5.0, "critical": 4.0, "urgent": 3.0,
            "high": 2.0, "medium": 1.0, "low": 0.5
        }


    def _calculate_score(self, server_id, start_time):
        """Calculates a composite score, incorporating load-balancing and dynamic procrastination penalties."""
        server = self.inst.servers[server_id]

        # 1. Compute Cost
        compute_cost = sum(self.inst.server_cost[server_id][t % self.inst.time_horizon]
                           for t in range(start_time, start_time + self.op.duration))

        # 2. Delay Penalty
        projected_completion_time = start_time + self.op.duration + self.remaining_duration
        delay = max(0, projected_completion_time - self.job.deadline)
        delay_cost = delay * self.job.delay_weight * self.inst.weights['delay_penalty']

        # 3. Dynamic Procrastination Penalty (Slack Crunch Penalty)
        # More sophisticated penalty for delaying, especially when close to deadline
        current_slack_for_op = self.job.deadline - projected_completion_time
        urgency_multiplier = self.priority_procrastination_multiplier.get(self.job.priority, 1.0)

        slack_ratio = 1.0
        if current_slack_for_op < 0: # Already past deadline, highly penalize
            slack_ratio = 1.0 + (math.exp(abs(current_slack_for_op) / (self.inst.time_horizon / 4.0)) - 1.0)
        elif current_slack_for_op < self.inst.time_horizon / 4: # Low slack, penalize quadratically
            ratio_to_threshold = (self.inst.time_horizon / 4.0 - current_slack_for_op) / (self.inst.time_horizon / 4.0)
            slack_ratio = 1.0 + (ratio_to_threshold**2) * 2.0

        procrastination_penalty = self.base_procrastination_penalty_factor \
                                * (start_time - self.earliest_start) \
                                * urgency_multiplier * slack_ratio

        # 4. Non-Linear Load-Balancing Penalty (Aggressive hotspot avoidance)
        load_penalty = 0.0
        if not server.is_cloud:
            # Lookahead window: current operation duration plus a bit more
            window_start = max(0, start_time - self.op.duration) # Look back slightly
            window_end = start_time + self.op.duration * 2 # Look ahead more aggressively
            total_utilization, num_timesteps = 0.0, 0
            for t in range(window_start, window_end):
                used = self.capacity_usage.get(server.id, {}).get(t, 0) # Use server.id not server_id directly
                available = get_available_capacity(self.inst, server.id, t) # Use server.id not server_id directly
                if available > 0:
                    total_utilization += used / available
                num_timesteps += 1
            if num_timesteps > 0:
                avg_utilization = total_utilization / num_timesteps
                load_prio_mult = self.priority_load_multiplier.get(self.job.priority, 1.0)
                # Using a squared penalty to heavily discourage scheduling on congested servers.
                load_penalty = (avg_utilization ** 2) * self.load_penalty_factor * load_prio_mult * self.job.delay_weight

        # Combine all cost components
        total_score = (compute_cost * self.inst.weights['compute_cost'] +
                       delay_cost + procrastination_penalty + load_penalty)

        return total_score

    def find_best_placement(self):
        """Implements the staged decision logic: benchmark cloud, search local, then decide."""

        best_overall_score = float('inf')
        chosen_server_id, chosen_start_time = None, None

        # 1. Evaluate the best cloud option (immediate start)
        cloud_placement_score = float('inf')
        best_cloud_server_id = None

        if compatible_cloud_servers:

            cheapest_cloud_server = min(compatible_cloud_servers, key=lambda s: self.inst.server_cost[s.id].get(self.earliest_start % TIME_HORIZON, float('inf')))
            cloud_placement_score = self._calculate_score(cheapest_cloud_server.id, self.earliest_start)
            best_cloud_server_id = cheapest_cloud_server.id

        # Initialize best overall with cloud if available
        if best_cloud_server_id:
            best_overall_score = cloud_placement_score
            chosen_server_id, chosen_start_time = best_cloud_server_id, self.earliest_start

        # 2. Search for the best local placement within a dynamic horizon
        # Calculate dynamic search horizon based on job priority and remaining slack
        total_rem_dur = self.op.duration + self.remaining_duration
        current_job_slack = self.job.deadline - (self.earliest_start + total_rem_dur)
        base_horizon = self.priority_base_horizon.get(self.job.priority, 64) # Default to medium if priority not found

        search_horizon = max(self.op.duration + 1, int(base_horizon * slack_multiplier))
        search_limit = self.earliest_start + search_horizon

        compatible_local_servers = [s for s in self.inst.servers.values() if s.server_type == self.op.server_type and not s.is_cloud]

        best_local_score, best_local_server_id, best_local_start_time = float('inf'), None, -1

        for server in compatible_local_servers:
            # Iterate through possible start times within the dynamic search horizon
            for start_t in range(self.earliest_start, search_limit + 1):
                if has_capacity(self.inst, self.capacity_usage, server.id, start_t, self.op.duration, self.op.capacity_required):
                    score = self._calculate_score(server.id, start_t)
                    if score < best_local_score:
                        best_local_score, best_local_server_id, best_local_start_time = score, server.id, start_t

        # 3. Strategic Decision: Compare best local found with best cloud
        cloud_pref_factor = self.cloud_aggressiveness_threshold.get(self.job.priority, 1.0)

        if best_local_server_id and best_local_score < (best_overall_score * cloud_pref_factor):
            # Local is significantly better (or within acceptable threshold for this priority)
            chosen_server_id, chosen_start_time = best_local_server_id, best_local_start_time
        elif chosen_server_id is None and best_local_server_id: # No cloud chosen yet, but a local option exists
             chosen_server_id, chosen_start_time = best_local_server_id, best_local_start_time
        # If cloud was initially the only or better option, it remains as chosen_server_id, chosen_start_time

        # 4. Failsafe: If no placement was found yet (e.g., no compatible cloud, no local in horizon)
        if chosen_server_id is None:
            if compatible_local_servers:
                # Unbounded search on the cheapest local machine as a last resort
                cheapest_server = min(compatible_local_servers, key=lambda s: sum(self.inst.server_cost[s.id].values()))
                start_t = self.earliest_start
                while True:
                    if has_capacity(self.inst, self.capacity_usage, cheapest_server.id, start_t, self.op.duration, self.op.capacity_required):
                        chosen_server_id, chosen_start_time = cheapest_server.id, start_t
                        break
                    start_t += 1
            else:
                raise Exception(f"Could not schedule op for job {self.job.id}, no compatible servers found.")

        return chosen_server_id, chosen_start_time


def schedule_job_online(inst: Instance,
                        job: Job,
                        schedule: Schedule,
                        capacity_usage: Dict[str, Dict[int, int]],
                        current_time: int):

    schedule[job.id] = []
    current_op_release_time = max(job.arrival_time, current_time)

    for idx, op in enumerate(job.operations):
        remaining_duration = sum(o.duration for o in job.operations[idx + 1:])

        evaluator = PlacementEvaluator(
            inst, job, op, capacity_usage, current_op_release_time, remaining_duration
        )
        chosen_server, chosen_start = evaluator.find_best_placement()

        # reserve capacity
        for t in range(chosen_start, chosen_start + op.duration):
            capacity_usage[chosen_server][t] = capacity_usage[chosen_server].get(t, 0) + op.capacity_required

        schedule[job.id].append({"server": chosen_server, "start": chosen_start})
        current_op_release_time = chosen_start + op.duration
